class FourierFeature(nn.Module):

        super().__init__()

        # 闅忔満棰戠巼鐭╅樀 B锛歴hape (in_c, out_c//2)锛屽厓绱?~ N(0, scale虏)
        fourier_basis = torch.randn(in_c, out_c // 2) * scale
        self.register_buffer('_fourier_basis', fourier_basis)

    def forward(self, x):

        # key-start
